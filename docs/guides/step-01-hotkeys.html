🔧 How It Works

1. pynput Listener (Background Thread)

self.listener = pkb.Listener(on_press=on_press, on_release=on_release)
self.listener.start()

# on_press: tracks which modifier keys are held
# on_release: removes them from the held set
# When a digit is pressed WHILE modifiers are held → _handle_combo(slot)

2. Modifier Tracking

self.held_mods = set()  # Tracks: ctrl_l, ctrl_r, alt_l, alt_r, ctrl, alt

# Left combo detection:
('ctrl' in mods and 'alt' in mods) or ('ctrl_l' in mods and 'alt_l' in mods)

# Right combo detection:
('ctrl_r' in mods or 'alt_r' in mods) and ('ctrl_r' in mods or 'ctrl' in mods) and ('alt_r' in mods or 'alt' in mods)

3. Digit → Slot Mapping

digit = key.char  # '1' through '9', '0'
slot = 10 if digit == '0' else int(digit)
# Result: keys 1-9 map to slots 1-9, key 0 maps to slot 10

def _register_hotkeys(self):
    def on_press(key):
        try:
            k = str(key).lower()
            if 'ctrl' in k:
                if 'left' in k or 'ctrl_l' in k:
                    self.held_mods.add('ctrl_l')
                elif 'right' in k or 'ctrl_r' in k:
                    self.held_mods.add('ctrl_r')
                else:
                    self.held_mods.add('ctrl')
            elif 'alt' in k:
                if 'left' in k or 'alt_l' in k:
                    self.held_mods.add('alt_l')
                elif 'right' in k or 'alt_r' in k:
                    self.held_mods.add('alt_r')
                else:
                    self.held_mods.add('alt')
            else:
                if hasattr(key, 'char') and key.char and key.char.isdigit():
                    digit = key.char
                    slot = 10 if digit == '0' else int(digit)
                    self._handle_combo(slot)
        except Exception as e:
            print(f"hotkey press err: {e}")

        # Removes modifiers from held_mods set
        ...

    self.listener = pkb.Listener(on_press=on_press, on_release=on_release)
    self.listener.start()

_handle_combo() — lines 635-644

def _handle_combo(self, slot: int):
    mods = self.held_mods
    has_right = ('ctrl_r' in mods or 'alt_r' in mods)

    if has_right and ('ctrl_r' in mods or 'ctrl' in mods) and ('alt_r' in mods or 'alt' in mods):
        print(f"[HOTKEY] Right combo → PASTE slot {slot}")
        self.paste_from_slot(slot)
    elif ('ctrl' in mods and 'alt' in mods) or ('ctrl_l' in mods and 'alt_l' in mods):
        print(f"[HOTKEY] Left combo → COPY slot {slot}")
        self.add_to_slot(slot)

📋 Copy Flow: add_to_slot()

def add_to_slot(self, slot_num: int):
    self._release_all_modifiers()      # Release any stuck keys
    pyautogui.hotkey("ctrl", "c")      # Trigger Ctrl+C in target app
    time.sleep(0.16)                    # Wait for clipboard
    content = pyperclip.paste()         # Read clipboard
    if content and content.strip():
        self.slots[str(slot_num)] = content
        self.save_slots()               # Persist to clipboard_dict.json
        self.show_toast(f"LEFT COMBO → COPY SLOT {slot_num:02d}", preview)

📋 Paste Flow: paste_from_slot()

def paste_from_slot(self, slot_num: int):
    content = self.slots.get(str(slot_num), "")
    if not content:
        return
    pyperclip.copy(content)             # Load clipboard
    time.sleep(0.12)
    self._release_all_modifiers()
    time.sleep(0.06)

    # xdotool FIRST (root-reliable)
    if self._is_terminal():
        subprocess.run(["xdotool", "key", "--clearmodifiers", "ctrl+shift+v"])
    else:
        subprocess.run(["xdotool", "key", "--clearmodifiers", "ctrl+v"])

🖥️ Terminal Detection

Paste uses ctrl+shift+v in terminals (most terminals use Shift), ctrl+v everywhere else:

def _is_terminal(self) -> bool:
    wid = subprocess.check_output(["xdotool", "getactivewindow"]).decode().strip()
    cls = subprocess.check_output(["xprop", "-id", wid, "WM_CLASS"]).decode().lower()
    return any(t in cls for t in ("terminal", "xterm", "konsole", "alacritty", "kitty", "foot"))

🔧 Two pynput Listeners Situation

There are TWO pynput listeners running simultaneously:

  1. MultiClipV2._register_hotkeys() — detects LCtrl+LAlt/RCtrl+RAlt for copy/paste
  2. HybridClipboardMonitor — detects ALL Ctrl+C for history capture

They coexist but are not ideal. Long-term debt: merge into one listener or use a callback from monitor to core.
