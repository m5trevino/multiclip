💾 Persistence

Snippets are saved to snippets.json in the project root:

{
  "0": "First snippet content",
  "1": "Second snippet content",
  "3": "Fourth snippet content"
}

Keys are 0-based indices. Empty snippets are NOT stored in the file.

Load on startup: _load_snippets() (line 649)
Save on edit: _save_snippet(idx) (line 664)

def _load_snippets(self):
    if os.path.exists(self.snippets_file):
        with open(self.snippets_file, 'r') as f:
            data = json.load(f)
            for k, v in data.items():
                idx = int(k)
                if idx in self.snippet_entries:
                    self.snippet_entries[idx].delete(0, 'end')
                    self.snippet_entries[idx].insert(0, v)

Save — lines 664-678

def _save_snippet(self, idx: int):
    content = self.snippet_entries[idx].get()
    data = {}
    if os.path.exists(self.snippets_file):
        with open(self.snippets_file, 'r') as f:
            data = json.load(f)
    data[str(idx)] = content
    with open(self.snippets_file, 'w') as f:
        json.dump(data, f, indent=2)
    self._toast(f"Snippet S{idx+1} saved")

📤 Send to Snippet

Transfer selected Clipman History items directly to snippets:

def _send_to_snippets(self, contents: list):
    sent = 0
    for content in contents:
        for i in range(8):
            entry = self.ui.snippet_entries.get(i)
            if entry and not entry.get().strip():
                entry.delete(0, 'end')
                entry.insert(0, content)
                self.ui._save_snippet(i)
                self.ui.flash_snippet(i, color="#32cd32")
                sent += 1
                break
    if sent:
        self.show_toast("Saved to Snippet", f"{sent} item(s) sent")
    else:
        messagebox.showwarning("Snippets Full", "All 8 slots occupied")

Behavior: Finds first empty snippet slot, fills it, saves to JSON, flashes green.

def _remove_snippet(self, idx: int):
    # Clear the entry
    self.snippet_entries[idx].delete(0, 'end')
    # Remove from JSON file
    data = {}
    if os.path.exists(self.snippets_file):
        with open(self.snippets_file, 'r') as f:
            data = json.load(f)
    data[str(idx)] = ""  # or: del data[str(idx)]
    with open(self.snippets_file, 'w') as f:
        json.dump(data, f, indent=2)
    self._toast(f"Snippet S{idx+1} removed")
